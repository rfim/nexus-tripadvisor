04 · The stream job · Python, with a Java twin

Event time, decided by the watermark

The watermark is the minimum over active partitions, less the out-of-orderness. A quiet partition stops holding it back; when every partition is quiet, processing time carries it.

wm = max(wm, minactive p max_event(p) − ooo) · fire ⟺ end ≤ wm · drop ⟺ end + lateness ≤ wm
src/nexus/stream.pyFlink semantics
    def _watermark(self, now: int) -> int:
        if self.wm_mode == "global_max":    # one clock for the topic: the newest event anywhere
            return max(self.s.wm, max(self.s.part_max.values()) - self.ooo) if self.s.part_max else self.s.wm
        if self.wm_mode == "min_all":       # every partition counts, quiet or not: a quiet one holds everything back
            return max(self.s.wm, min(self.s.part_max.values()) - self.ooo) if self.s.part_max else self.s.wm
        active = [self.s.part_max[p] for p, seen in self.s.part_seen.items() if now - seen < IDLE]
        if active:
            wm = min(active) - self.ooo
        elif self.s.part_max:           # every partition idle: let processing time carry the watermark (a heartbeat)
            wm = now - self.ooo - IDLE
        else:
            wm = NEG
        return max(self.s.wm, wm)

    def _emit(self, key, w: dict, now: int):
        value = {"n": w["n"], "sum": round(w["sum"], 4)}
        old = self.sink.get(key)
        self.sink[key] = {"value": value, "first_emit": old["first_emit"] if old else now, "last_emit": now,
                          "emits": (old["emits"] if old else 0) + 1}
        self.emits.append((now, key[0] + self.size))

    def _advance(self, now: int):
        wm = self._watermark(now)
        if wm <= self.s.wm and self.s.wm != NEG:
            return
        self.s.wm = wm

The rules

Fire when the watermark passes a window's end; re-fire on a late record within the hour; past it, the side output.

Dedupe on event_id, kept 24 hours. Dead letters for what can't be read, with the reason.

Offsets, windows, dedupe state and the ledger are checkpointed together; sinks are keyed, so a restore re-sends without changing anything.

def recompute(spark: SparkSession, name: str) -> dict:
    t = topic(name)
    prod = t["product"]
    size = prod["window_minutes"] * 60_000
    raw = (spark.read.json(str(LANDING / "kafka" / name / "p*.jsonl"))
           .withColumn("partition", F.regexp_extract(F.input_file_name(), r"p(\d+)\.jsonl", 1).cast("int"))
           .withColumn("v", F.from_json("value", VALUE)))
    rates = F.create_map(*[x for k, r in fx().items() for x in (F.lit(k), F.lit(r))])
    parsed = (raw.withColumn("ok", F.col("v.event_time").isNotNull())
              .withColumn("id", F.col(f"v.{t['id']}"))
              .withColumn("et", F.unix_millis(F.to_timestamp("v.event_time", "yyyy-MM-dd'T'HH:mm:ss.SSSX")))
              .withColumn("revenue_gbp", F.coalesce(F.col("v.cpc_gbp"),
                                                    F.round(F.col("v.price.amount") * rates[F.col("v.price.currency")], 4))))
    bad = parsed.filter(~F.col("ok")).count()
    first = Window.partitionBy("id").orderBy("ts", "partition", "offset")
    events = (parsed.filter("ok").withColumn("rn", F.row_number().over(first)).filter("rn = 1")
              .withColumn("window_start", F.col("et") - F.pmod(F.col("et"), F.lit(size))))
    group = [F.col(f"v.{g}").alias(g) for g in prod["group_by"]]
    measure = F.sum("revenue_gbp") if any(h == "sum" for h in prod["measures"].values()) else F.lit(0.0)
    rows = events.groupBy("window_start", *group).agg(F.count("*").alias("n"), measure.alias("sum")).collect()
    return {"windows": {(r["window_start"], tuple(r[g] for g in prod["group_by"])): {"n": r["n"], "sum": round(r["sum"] or 0.0, 4)}
                        for r in rows},
            "bad_records": bad, "events": events.count()}

06 · Modeling · SQL from the contract

One MERGE, generated for both warehouses

The key from the contract becomes the MERGE condition; a late re-fire updates one row; a tombstone deletes.

src/nexus/warehouse.pyrenders BigQuery + Snowflake
def _merge(d: str, table: str, spec: dict) -> str:
    cols, key = list(spec["cols"]), spec["key"]
    tgt, src = _name(d, "gold", table), _name(d, "stage", table)
    now = "CURRENT_TIMESTAMP()"
    on = " AND ".join(f"t.{k} = s.{k}" for k in key)
    vals = [c for c in cols if c not in key]
    changed = " OR ".join(f"t.{c} IS DISTINCT FROM s.{c}" for c in vals)
    sets = ", ".join(f"{c} = s.{c}" for c in vals)
    ins = ", ".join(cols)
    sql = f"MERGE INTO {tgt} t\nUSING {src} s\n  ON {on}\n"
    if "version" in spec:
        v, tomb = spec["version"], spec["tombstone"]
        sql += (f"WHEN MATCHED AND s.{tomb} AND s.{v} > t.{v} THEN DELETE\n"
                f"WHEN MATCHED AND NOT s.{tomb} AND s.{v} > t.{v} THEN UPDATE SET {sets}, _loaded_at = {now}\n"
                f"WHEN NOT MATCHED AND NOT s.{tomb} THEN INSERT ({ins}, _loaded_at)\n  VALUES ({', '.join('s.' + c for c in cols)}, {now});\n")
    else:
        sql += (f"WHEN MATCHED AND ({changed}) THEN UPDATE SET {sets}, _loaded_at = {now}\n"
                f"WHEN NOT MATCHED THEN INSERT ({ins}, _loaded_at)\n  VALUES ({', '.join('s.' + c for c in cols)}, {now});\n")
    return sql

07 · SLAs · the monitor

Freshness, measured like a consumer feels it

Age is now minus the completeness the job publishes. Sampled every five minutes; the incident record is built from the same samples.

src/nexus/sla.pyPython
def age_series(job, start: int, end: int) -> list[tuple[int, float]]:
    emits = sorted(job.progress)
    out, i, visible = [], 0, None
    for t in _samples(start, end):
        while i < len(emits) and emits[i][0] <= t:
            visible = emits[i][1] if visible is None else max(visible, emits[i][1])
            i += 1
        if visible is not None:
            out.append((t, round((t - visible) / MIN, 1)))
    return out


def lag_series(job, start: int, end: int) -> list[tuple[int, float]]:
    """Minutes the oldest record that has arrived but isn't processed yet has been waiting (0 when caught up)."""
    recs = sorted(zip((r.ts for r in job.records), job.done_at))
    out, i, waiting = [], 0, []
    for t in _samples(start, end):
        while i < len(recs) and recs[i][0] <= t:
            heapq.heappush(waiting, recs[i])
            i += 1
        while waiting and waiting[0][1] <= t:
            heapq.heappop(waiting)
        # a record processed later than an older one still waiting stays in the heap: drop finished ones lazily
        live = [w for w in waiting if w[1] > t]
        out.append((t, round((t - min(w[0] for w in live)) / MIN, 1) if live else 0.0))
    return out

The same job, on Flink

Parse with a dead-letter output, dedupe in keyed state, per-partition watermarks, event-time windows with allowed lateness, a side output for what's too late.

records → Parse ⟶ DLQ · keyBy(id) → Dedupe(TTL 24 h) · keyBy(group) → Tumbling(1 h) → allowedLateness(1 h) ⟶ late
flink/job/.../NexusJobs.javaFlink 1.20.1 · Java 17
    public static Outputs build(DataStream<KafkaRecord> records, Product p, boolean replayWatermarks) {
        SingleOutputStreamOperator<Event> parsed = records.process(new Parse(p.topic())).name("parse " + p.topic());
        SingleOutputStreamOperator<Event> unique = parsed.keyBy(e -> e.id).process(new Dedupe()).name("dedupe on id");
        final long ooo = p.outOfOrderness().toMillis();       // a long, not the record: the lambda is serialized
        DataStream<Event> timed = replayWatermarks
                ? unique.assignTimestampsAndWatermarks(WatermarkStrategy
                        .<Event>forGenerator(ctx -> new PartitionWatermarks(ooo))
                        .withTimestampAssigner((e, ts) -> e.eventTime)).name("watermarks")
                : unique;
        SingleOutputStreamOperator<WindowRow> windows = timed
                .keyBy(e -> e.group)
                .window(TumblingEventTimeWindows.of(p.window()))
                .allowedLateness(p.lateness())
                .sideOutputLateData(TOO_LATE)
                .aggregate(new Windows.CountSum(), new Windows.Emit())
                .name(p.name());
        return new Outputs(windows, parsed.getSideOutput(Parse.DEAD_LETTERS), unique.getSideOutput(Dedupe.DUPLICATES),
                windows.getSideOutput(TOO_LATE));
    }
